```{if:{argslen}|=|0|then:{code:.t {tagname} <effects> <ppm_name>}|else:
{if:{arg:0}|=|help|then:{code:moviecommand effects:{newline}ccshue=VALUE{newline}hsladjust=VALUE{newline}invertlum{newline}rgb=rgb{newline}rgb=rbg{newline}rgb=grb{newline}rgb=gbr{newline}rgb=brg{newline}rgb=bgr{newline}negate{newline}{newline}Usage:{newline}.t moviecommand effect1,effect2,effect3 [ppm_name]{newline}{newline}Example:{newline}.t moviecommand ccshue=180,hsladjust=0.3,invertlum,rgb=bgr,negate movie}|else:{set:code|{js:
let effects = `{arg:0}`;
let ppm = `{or:{arg:1}|movie}`;
let code = "";
let error = "";

for (let effect of effects.split(",")) {

if (effect.startsWith("ccshue=")) {
let angle = effect.replace("ccshue=","");
code += '-colorspace yuv -fx "angle=' + angle + '*pi/180; channel(u,.5+(u.g-.5)*cos(angle)-(u.b-.5)*sin(angle),.5+(u.g-.5)*sin(angle)+(u.b-.5)*cos(angle))" -colorspace srgb ';
}

else if (effect.startsWith("hsladjust=")) {
let value = effect.replace("hsladjust=","");
code += '-define modulate:colorspace=hsl -modulate 100,100,%[fx:' + value + '*200+100] ';
}

else if (effect === "invertlum") {
code += '-define modulate:colorspace=hsl -colorspace lab -channel r -level -5%,95% -negate +channel -colorspace srgb ';
}

else if (effect === "rgb=bgr") {
code += '-color-matrix "0 0 1 0 1 0 1 0 0" ';
}

else if (effect === "rgb=rbg") {
code += '-color-matrix "1 0 0 0 0 1 0 1 0" ';
}

else if (effect === "rgb=grb") {
code += '-color-matrix "0 1 0 1 0 0 0 0 1" ';
}

else if (effect === "rgb=gbr") {
code += '-color-matrix "0 1 0 0 0 1 1 0 0" ';
}

else if (effect === "rgb=brg") {
code += '-color-matrix "0 0 1 1 0 0 0 1 0" ';
}

else if (effect === "rgb=rgb") {
code += '-color-matrix "1 0 0 0 1 0 0 0 1" ';
}

else if (effect === "negate") {
code += '-negate ';
}

else {
console.error("Command " + effect + " does not exist. Use `.t moviecommand help` to view all working commands.");
break;
}

}

if (error !== "") {
console.error(error);
} else {
console.log('-vf "movie=$(magick hald:6 ' + code + ppm + '.ppm; echo ' + ppm + '.ppm),[in]haldclut",format=yuv444p');
}
}} {code:{get:code}} {sh:
load {iv}

ffmpeg -i $FILE_1 {get:code} ./output/$(basename movie.${FILE_1##*.})}}}}
```
